import pygame

# ===================
# Configurações
# ===================

LARGURA_TELA = 800
ALTURA_TELA = 600

FPS = 60

# CORES

VERDE_GRAMA = (90, 180, 80)
MARROM_TERRA = (150, 100, 50)
AZUL_CEU = (100, 180, 240)
BRANCO = (255, 255, 255)


# ============================
# JOGADOR
# ============================

TAMANHO_JOGADOR = 30
VELOCIDADE = 4


def criar_jogador():

    return {
        "x": 400,
        "y": 300
    }


def mover_jogador(jogador, teclas):

    # Esquerda
    if teclas[pygame.K_LEFT]:
        jogador["x"] -= VELOCIDADE

    # Direita
    if teclas[pygame.K_RIGHT]:
        jogador["x"] += VELOCIDADE

    # Cima
    if teclas[pygame.K_UP]:
        jogador["y"] -= VELOCIDADE

    # Baixo
    if teclas[pygame.K_DOWN]:
        jogador["y"] += VELOCIDADE


# ======================================
# DESENHAR JOGADOR
# ======================================

def desenhar_jogador(tela, jogador):

    jogador_rect = pygame.Rect(
        jogador["x"],
        jogador["y"],
        TAMANHO_JOGADOR,
        TAMANHO_JOGADOR
    )

    pygame.draw.rect(
        tela,
        BRANCO,
        jogador_rect
    )


# ======================================
# DESENHAR FAZENDA
# ======================================

def desenhar_fazenda(tela):

    # Fundo da fazenda

    tela.fill(VERDE_GRAMA)


# ======================================
# PROGRAMA PRINCIPAL
# ======================================

def main():

    pygame.init()

    # Criar a janela

    tela = pygame.display.set_mode(
        (LARGURA_TELA, ALTURA_TELA)
    )

    pygame.display.set_caption(
        "Minha Fazenda"
    )

    relogio = pygame.time.Clock()

    jogador = criar_jogador()

    rodando = True

    while rodando:

        # -----------------------
        # EVENTOS
        # -----------------------

        for evento in pygame.event.get():

            if evento.type == pygame.QUIT:

                rodando = False


        # -----------------------
        # TECLAS
        # -----------------------

        teclas = pygame.key.get_pressed()


        # -----------------------
        # MOVIMENTO
        # -----------------------

        mover_jogador(
            jogador,
            teclas
        )


        # -----------------------
        # DESENHAR
        # -----------------------

        desenhar_fazenda(
            tela
        )

        desenhar_jogador(
            tela,
            jogador
        )

        pygame.display.flip()


        # -----------------------
        # FPS
        # -----------------------

        relogio.tick(FPS)


    pygame.quit()


# ======================================
# INICIAR JOGO
# ======================================

if __name__ == "__main__":

    main()

def plantar_semente(jogador, terreno):

    jogador_rect = pygame.Rect(
        jogador["x"],
        jogador["y"],
        TAMANHO_JOGADOR,
        TAMANHO_JOGADOR
    )

    for quadrado in terreno:

        x = TERRENO_X + (
            quadrado["coluna"] * TAMANHO_TERRENO
        )

        y = TERRENO_Y + (
            quadrado["linha"] * TAMANHO_TERRENO
        )

        terreno_rect = pygame.Rect(
            x,
            y,
            TAMANHO_TERRENO,
            TAMANHO_TERRENO
        )

        # Distância horizontal

        distancia_x = (
            jogador_rect.centerx -
            terreno_rect.centerx
        )

        # Distância vertical

        distancia_y = (
            jogador_rect.centery -
            terreno_rect.centery
        )

        # Distância total

        distancia_total = (
            distancia_x ** 2 +
            distancia_y ** 2
        ) ** 0.5

        if distancia_total <= DISTANCIA_INTERACAO:

            quadrado["plantado"] = True

            break

Parte 4

Nesta etapa, vamos melhorar a plantação. Em vez de a planta aparecer imediatamente como uma bolinha verde, ela terá estágios de crescimento:

vazio → ainda não plantou nada
semente → acabou de plantar
crescendo → a planta está crescendo
pronta → pode ser colhida

# DISTÂNCIA ENTRE JOGADOR E TERRENO

def esta_perto(jogador, quadrado):

    jogador_rect = pygame.Rect(

        jogador["x"],

        jogador["y"],

        TAMANHO_JOGADOR,

        TAMANHO_JOGADOR

    )


    x = TERRENO_X + (

        quadrado["coluna"] * TAMANHO_TERRENO

    )

    y = TERRENO_Y + (

        quadrado["linha"] * TAMANHO_TERRENO

    )


    terreno_rect = pygame.Rect(

        x,

        y,

        TAMANHO_TERRENO,

        TAMANHO_TERRENO

    )


    distancia_x = (

        jogador_rect.centerx -

        terreno_rect.centerx

    )


    distancia_y = (

        jogador_rect.centery -

        terreno_rect.centery

    )


    distancia_total = (

        distancia_x ** 2 +

def plantar_semente(jogador, terreno):

    for quadrado in terreno:

        if esta_perto(jogador, quadrado):

            if quadrado["estado"] == "vazio":

                quadrado["estado"] = "semente"

def crescer_planta(jogador, terreno):

    for quadrado in terreno:

        if esta_perto(jogador, quadrado):

            if quadrado["estado"] == "semente":

                quadrado["estado"] = "crescendo"

                print("A planta começou a crescer!")

                break


            elif quadrado["estado"] == "crescendo":

                quadrado["estado"] = "pronta"

                print("A planta está pronta para colher!")

                break


# ======================================

def colher_planta(jogador, terreno):

    for quadrado in terreno:

                quadrado["estado"] = "vazio"

                print("Planta colhida!")
